  name = "${var.environment}-worker-policy"

  policy = jsonencode({
    Version = "2012-10-17"

    Statement = [
      {
        Sid    = "ObjectAccess"
        Effect = "Allow"

        Action = [
          "s3:GetObject",
          "s3:PutObject",
          "s3:DeleteObject"
        ]

        # Vulnerability: object actions should only access objects
        # belonging to the storage module's bucket.
        Resource = "*"
      },

      {
        Sid    = "BucketAccess"
        Effect = "Allow"

        Action = [
          "s3:ListBucket"
        ]

        Resource = local.resource_scope.bucket_path
      },

      {
        Sid    = "EncryptionAccess"
        Effect = "Allow"

        Action = [
          "kms:Encrypt",
          "kms:Decrypt",
          "kms:GenerateDataKey"
        ]

        # Vulnerability: KMS actions should only access the key
        # used by the storage module.
        Resource = "*"
      }
    ]
  })
}
